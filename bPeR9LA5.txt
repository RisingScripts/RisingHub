local function waitForOwnedBandana(hatsFrame)
    local owned = hatsFrame:FindFirstChild("Blue Head Bandana")
    if owned then
        return owned
    end

    repeat
        RunService.Heartbeat:Wait()
        owned = hatsFrame:FindFirstChild("Blue Head Bandana")
    until owned or not infiniteMoneyEnabled

    return owned
end

local function infiniteMoneyLoop()
    local success = pcall(function()
        if not LocalPlayer or not replicatesignal then
            return
        end

        local ui = getInfiniteMoneyGuiElements()
        local hatsFrame = ui.HatsFrame

        if not hatsFrame:FindFirstChild("Blue Head Bandana") then
            replicatesignal(ui.ShopBandana.MouseButton1Down, 0, 0)
            RunService.Heartbeat:Wait()
            replicatesignal(ui.PurchaseButton.MouseButton1Down, 0, 0)
        end

        local blueBandana = waitForOwnedBandana(hatsFrame)
        if not blueBandana then
            return
        end

        while infiniteMoneyEnabled do
            replicatesignal(blueBandana.MouseButton1Down, 0, 0)
            RunService.Heartbeat:Wait()

            replicatesignal(ui.WearButton.MouseButton1Down, 0, 0)
            RunService.Heartbeat:Wait()

            replicatesignal(ui.NoneButton.MouseButton1Down, 0, 0)
            RunService.Heartbeat:Wait()

            replicatesignal(ui.PurchaseButton.MouseButton1Down, 0, 0)
            RunService.Heartbeat:Wait()
        end
    end)

    if not success and infiniteMoneyEnabled then
        task.wait(1)
    end
end

local function highlightTarget(target)
    if not highlightEnabled then return end
    if target and target.Character then
        local hl = target.Character:FindFirstChild("AimbotHighlight")
        if not hl then
            hl = Instance.new("Highlight")
            hl.Name = "AimbotHighlight"
            hl.FillColor = Color3.fromRGB(255, 0, 0)

local function getClosestPlayerToMouse()
    local mousePos = UserInputService:GetMouseLocation()
    local minDist = fovRadius
    local closestChar = nil
    for _, v in ipairs(Players:GetPlayers()) do
        if v ~= LocalPlayer and v.Character and v.Character:FindFirstChild("Head") and v.Character:FindFirstChild("Humanoid") and v.Character.Humanoid.Health > 0 then
            if not teamCheck or v.Team ~= LocalPlayer.Team then
                local pos, onscreen = Camera:WorldToViewportPoint(v.Character.Head.Position)
                if onscreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(mousePos.X, mousePos.Y)).Magnitude
                    if dist < minDist then
                        minDist = dist
                        closestChar = v.Character
                    end
                end
            end
        end
    end
    return closestChar
end

local function GetUsernames()
    local character = localPlayer and localPlayer.Character
    if not character then return nil end
    local head = character:FindFirstChild("Head")
    if not head then return nil end
    local displayGUI = head:FindFirstChild("displayGUI")
    if not displayGUI then return nil end
    local nameLabel = displayGUI:FindFirstChild("nameLabel")
    local ownerLabel = displayGUI:FindFirstChild("ownerLabel")
    return nameLabel, ownerLabel
end

local function GetAllUsernames()
    local Players = game:GetService("Players")
    local allLabels = {}
    for _, player in ipairs(Players:GetPlayers()) do
        local character = player.Character
        if character then
            local head = character:FindFirstChild("Head")
            if head then
                local displayGUI = head:FindFirstChild("displayGUI")
                if displayGUI then
                    local nameLabel = displayGUI:FindFirstChild("nameLabel")
                    local ownerLabel = displayGUI:FindFirstChild("ownerLabel")
                    table.insert(allLabels, {nameLabel = nameLabel, ownerLabel = ownerLabel})
                end
            end
        end
    end
    return allLabels
end

local function ResetDirections()
    for k in pairs(FlyFeature.Direction) do FlyFeature.Direction[k] = 0 end
end

local velocityHandlerName = "PHFly_Position"
local gyroHandlerName = "PHFly_Gyro"
local mfly1, mfly2
local FLYING = false

local function getRoot(char)
    return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
end

local function unmobilefly(speaker)
    pcall(function()
        FLYING = false
        local root = getRoot(speaker.Character)
        if root and root:FindFirstChild(velocityHandlerName) then
            root:FindFirstChild(velocityHandlerName):Destroy()
        end
        if root and root:FindFirstChild(gyroHandlerName) then
            root:FindFirstChild(gyroHandlerName):Destroy()
        end
        local hum = speaker.Character and speaker.Character:FindFirstChildWhichIsA("Humanoid")
        if hum then
            hum.PlatformStand = false
        end
        if mfly1 then mfly1:Disconnect() end
        if mfly2 then mfly2:Disconnect() end
    end)
end

local function mobilefly(speaker, vfly)
    unmobilefly(speaker)
    FLYING = true

    local root = getRoot(speaker.Character)
    local camera = workspace.CurrentCamera
    local v3none = Vector3.new()
    local v3zero = Vector3.new(0, 0, 0)
    local v3inf = Vector3.new(9e9, 9e9, 9e9)

    local controlModule
    pcall(function()
        controlModule = require(speaker:FindFirstChild("PlayerScripts") and speaker.PlayerScripts:WaitForChild("PlayerModule"):WaitForChild("ControlModule"))
    end)
    if not (controlModule and root) then return end

    local bv = Instance.new("BodyVelocity")
    bv.Name = velocityHandlerName
    bv.Parent = root
    bv.MaxForce = v3zero
    bv.Velocity = v3zero

    local bg = Instance.new("BodyGyro")
    bg.Name = gyroHandlerName
    bg.Parent = root
    bg.MaxTorque = v3inf
    bg.P = 1000
    bg.D = 50

    mfly1 = speaker.CharacterAdded:Connect(function()
        local root2 = getRoot(speaker.Character)
        if root2 then
            local bv2 = Instance.new("BodyVelocity")
            bv2.Name = velocityHandlerName
            bv2.Parent = root2
            bv2.MaxForce = v3zero
            bv2.Velocity = v3zero

            local bg2 = Instance.new("BodyGyro")
            bg2.Name = gyroHandlerName
            bg2.Parent = root2
            bg2.MaxTorque = v3inf
            bg2.P = 1000
            bg2.D = 50
        end
    end)

    mfly2 = RunService.RenderStepped:Connect(function()
        root = getRoot(speaker.Character)
        camera = workspace.CurrentCamera
        if not (speaker.Character and root and root:FindFirstChild(velocityHandlerName) and root:FindFirstChild(gyroHandlerName)) then return end
        local humanoid = speaker.Character:FindFirstChildWhichIsA("Humanoid")
        local VelocityHandler = root:FindFirstChild(velocityHandlerName)
        local GyroHandler = root:FindFirstChild(gyroHandlerName)
        if not (humanoid and VelocityHandler and GyroHandler) then return end

        VelocityHandler.MaxForce = v3inf
        GyroHandler.MaxTorque = v3inf
        if not vfly then humanoid.PlatformStand = true end
        GyroHandler.CFrame = camera.CFrame
        VelocityHandler.Velocity = v3none

        local direction = controlModule:GetMoveVector()
        local flyspeed = FlyFeature.Speed or FlySpeedValue or getgenv().FlySpeedValue or 50
        if direction.X ~= 0 then
            VelocityHandler.Velocity = VelocityHandler.Velocity + camera.CFrame.RightVector * (direction.X * flyspeed * 5)
        end
        if direction.Z ~= 0 then
            VelocityHandler.Velocity = VelocityHandler.Velocity - camera.CFrame.LookVector * (direction.Z * flyspeed * 5)
        end
    end)
end

local function DisableFly()
    FlyFeature.Enabled = false
    if FlyFeature.Conn_RenderStep then
        pcall(function() FlyFeature.Conn_RenderStep:Disconnect() end)
        FlyFeature.Conn_RenderStep = nil
    end
    DisconnectInputs()
    local char = LocalPlayer and LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    RemoveBodyMovers(root)
    local hum = char and char:FindFirstChildWhichIsA("Humanoid")
    if hum then hum.PlatformStand = false end
    ResetDirections()
    if IsMobile() then
        unmobilefly(LocalPlayer)
    end
end

local function EnableFly()
    DisableFly()
    FlyFeature.Enabled = true

    local char = LocalPlayer and LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildWhichIsA("Humanoid")
    if not (char and root and hum) then return DisableFly() end

    if not IsMobile() then
        RemoveBodyMovers(root)
        local bp = Instance.new("BodyPosition")
        bp.Name = "PHFly_Position"
        bp.MaxForce = Vector3.new(1e5,1e5,1e5)
        bp.D = 10
        bp.P = 30000
        bp.Position = root.Position
        bp.Parent = root

        local bg = Instance.new("BodyGyro")
        bg.Name = "PHFly_Gyro"
        bg.MaxTorque = Vector3.new(1e5,1e5,1e5)
        bg.P = 15000
        bg.CFrame = root.CFrame
        bg.Parent = root

        ResetDirections()

        table.insert(FlyFeature.Conns_Input, UserInputService.InputBegan:Connect(function(input, gpe)
            if gpe or not FlyFeature.Enabled then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                local key = input.KeyCode.Name
                if directionsByKey[key] then
                    FlyFeature.Direction[directionsByKey[key]] = 1
                end
            end
        end))
        table.insert(FlyFeature.Conns_Input, UserInputService.InputEnded:Connect(function(input, gpe)
            if gpe or not FlyFeature.Enabled then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                local key = input.KeyCode.Name
                if directionsByKey[key] then
                    FlyFeature.Direction[directionsByKey[key]] = 0
                end
            end
        end))
        FlyFeature.Conn_RenderStep = RunService.RenderStepped:Connect(function(dt)
            if not (FlyFeature.Enabled and root and hum and bp and bg) then
                DisableFly()
                return
            end
            local cam = workspace.CurrentCamera
            local cf = cam and cam.CFrame or root.CFrame
            local move = Vector3.new()

            if FlyFeature.Direction.F == 1 then move = move + cf.LookVector end
            if FlyFeature.Direction.B == 1 then move = move - cf.LookVector end
            if FlyFeature.Direction.R == 1 then move = move + cf.RightVector end
            if FlyFeature.Direction.L == 1 then move = move - cf.RightVector end
            if FlyFeature.Direction.U == 1 then move = move + Vector3.new(0,1,0) end
            if FlyFeature.Direction.D == 1 then move = move - Vector3.new(0,1,0) end

            if move.Magnitude > 0 then
                move = move.Unit * (FlyFeature.Speed or 50)
            end

            bp.Position = root.Position + move * dt
            bg.CFrame = cf
            hum.PlatformStand = true
        end)
    else
        mobilefly(LocalPlayer, false)
    end
end

local function getAllGuns()
    local guns = {}
    local function scan(container)
        for _, tool in ipairs(container:GetChildren()) do

                table.insert(guns, tool)

local function applyGunMods(gun, mods)
    local settings = gun:FindFirstChild("Settings")
    if not settings or not settings:IsA("ModuleScript") then return end
    local gunData = require(settings)
    if type(gunData) ~= "table" then return end

    if mods.infDamage then
        if gunData.headDamage then gunData.headDamage = math.huge end
        if gunData.torsoDamage then gunData.torsoDamage = math.huge end
        if gunData.armDamage then gunData.armDamage = math.huge end
        if gunData.legDamage then gunData.legDamage = math.huge end
        if gunData.handDamage then gunData.handDamage = math.huge end
        if gunData.footDamage then gunData.footDamage = math.huge end
        if gunData.vestDamage then gunData.vestDamage = math.huge end
        if gunData.hatDamage then gunData.hatDamage = math.huge end
    end
    if mods.fastReload then
        if gunData.reloadAnimation then gunData.reloadAnimation = 0 end
        if gunData.reloadSound then gunData.reloadSound = 0 end
    end
    if mods.antiJam then
        if gunData.jamChance then gunData.jamChance = 0 end
        if gunData.jamIncrease then gunData.jamIncrease = 0 end
    end
    if mods.autoActivate then
        if gunData.fireMode then gunData.fireMode = "auto" end
        if gunData.autoCooldown then gunData.autoCooldown = 0.01 end
    end
    if mods.noRecoil then
        if gunData.recoil then gunData.recoil = 0 end
        if gunData.spread then gunData.spread = 0 end
    end
    if mods.fastEquip then
        if gunData.equipAnimation then gunData.equipAnimation = 0 end
        if gunData.unequipAnimation then gunData.unequipAnimation = 0 end
        if gunData.sprintAnimation then gunData.sprintAnimation = 0 end
        if gunData.raiseAnimation then gunData.raiseAnimation = 0 end
        if gunData.pointAnimation then gunData.pointAnimation = 0 end
    end
    if mods.fastShoot then
        if gunData.semiCooldown then gunData.semiCooldown = 0 end
        if gunData.autoCooldown then gunData.autoCooldown = 0 end
    end
end

local function applyAllMods()
    for _, gun in ipairs(getAllGuns()) do
        applyGunMods(gun, modState)
    end
end

function createBoxEsp(v)
    local Box = Drawing.new("Square")
    Box.Visible = false
    Box.Filled = false

    local HealthBar = Drawing.new("Line")
    HealthBar.Visible = false
    HealthBar.Thickness = 2

    local NameTag = Drawing.new("Text")
    NameTag.Visible = false
    NameTag.Size = 9
    NameTag.Center = true

    local DistanceTag = Drawing.new("Text")
    DistanceTag.Visible = false
    DistanceTag.Size = 9
    DistanceTag.Center = true

    local Tracer = Drawing.new("Line")
    Tracer.Visible = false

    local TracerOutline = Drawing.new("Line")
    TracerOutline.Visible = false
    TracerOutline.Color = Color3.new(0, 0, 0)

    playerDrawings[v] = {
        Box = Box,
        Health = HealthBar,
        Name = NameTag,
        Distance = DistanceTag,
        Tracer = Tracer,
        TracerOutline = TracerOutline
    }

    game:GetService("RunService").RenderStepped:Connect(function()
        if not ESP.enabled then
            for _, d in pairs(playerDrawings[v]) do
                d.Visible = false
            end
            return
        end

        if v and v.Character and v.Character:FindFirstChild("HumanoidRootPart") and v.Character:FindFirstChild("Humanoid") and v.Character:FindFirstChild("Head") and v ~= lplr and v.Character.Humanoid.Health > 0 then
            if ESP.teamcheck and v.Team == lplr.Team then
                for _, d in pairs(playerDrawings[v]) do
                    d.Visible = false
                end
                return
            end

            local Root = v.Character.HumanoidRootPart
            local Head = v.Character.Head
            local RootPos, onScreen = camera:WorldToViewportPoint(Root.Position)
            local HeadPos = camera:WorldToViewportPoint(Head.Position + HeadOff)
            local LegPos = camera:WorldToViewportPoint(Root.Position - LegOff)

            local boxSize = Vector2.new(1000 / RootPos.Z, HeadPos.Y - LegPos.Y)
            local boxPos = Vector2.new(RootPos.X - boxSize.X / 2, RootPos.Y - boxSize.Y / 2)

            Box.Position = boxPos
            Box.Size = boxSize
            Box.Filled = ESP.box_filled
            Box.Color = ESP.team_boxes[2]
            Box.Thickness = ESP.box_outline and ESP.team_boxes[3] or 0
            Box.Transparency = ESP.team_boxes[4]
            Box.Visible = ESP.team_boxes[1] and onScreen

            if ESP.team_health then
                local hp = v.Character.Humanoid.Health / v.Character.Humanoid.MaxHealth
                HealthBar.From = Vector2.new(boxPos.X + boxSize.X + 5, boxPos.Y + boxSize.Y * (1 - hp))
                HealthBar.To = Vector2.new(boxPos.X + boxSize.X + 5, boxPos.Y + boxSize.Y)
                HealthBar.Color = Color3.new(1 - hp, hp, 0)
                HealthBar.Transparency = ESP.health_transparency
                HealthBar.Visible = onScreen
            else
                HealthBar.Visible = false
            end

            NameTag.Text = v.Name
            NameTag.Position = Vector2.new(boxPos.X + boxSize.X / 2, boxPos.Y - 30)  
            NameTag.Color = ESP.team_names[2]
            NameTag.Outline = ESP.outlines
            NameTag.OutlineColor = Color3.new(0, 0, 0)
            NameTag.Transparency = ESP.team_names[3]
            NameTag.Visible = ESP.team_names[1] and onScreen
            NameTag.Size = 24 
            NameTag.Font = Drawing.Fonts.Plex 

            local dist = (lplr.Character.HumanoidRootPart.Position - Root.Position).Magnitude
            DistanceTag.Text = string.format("%dm", math.floor(dist))
            DistanceTag.Position = Vector2.new(boxPos.X + boxSize.X / 2, boxPos.Y + boxSize.Y + 1)
            DistanceTag.Color = Color3.fromRGB(255, 255, 255)
            DistanceTag.Outline = ESP.outlines
            DistanceTag.OutlineColor = Color3.new(0, 0, 0)
            DistanceTag.Transparency = ESP.distance_transparency
            DistanceTag.Visible = ESP.team_distance and onScreen
            DistanceTag.Size = 24
            DistanceTag.Font = Drawing.Fonts.Plex 

            local from = camera:WorldToViewportPoint(lplr.Character.Head.Position)
            Tracer.From = Vector2.new(from.X, from.Y)
            Tracer.To = Vector2.new(RootPos.X, RootPos.Y)
            Tracer.Color = ESP.team_tracer[2]
            Tracer.Thickness = ESP.team_tracer[3]
            Tracer.Transparency = ESP.team_tracer[4]

            TracerOutline.To = Tracer.To
            TracerOutline.Thickness = ESP.team_tracer[3] + 2
            TracerOutline.Visible = ESP.team_tracer[1] and ESP.tracer_outline and onScreen
        else
            for _, d in pairs(playerDrawings[v]) do
                d.Visible = false
            end
        end
    end)

local function round(n)
    return math.floor(n + 0.5)
end

local function getAlignedPosition(pos)
    return Vector3.new(
        round(pos.X) - round(pos.X)%4 + 2,
        round(pos.Y) - round(pos.Y)%4 + 2,
        round(pos.Z) - round(pos.Z)%4 + 2
    )
end

local function removeWater(voxelPos, worldSize)
    local Terrain = workspace.Terrain
    local processBlockSize = 640
    local WATER, AIR = Enum.Material.Water, Enum.Material.Air
    voxelPos = getAlignedPosition(voxelPos)
    local region = Region3.new(voxelPos - worldSize/2, voxelPos + worldSize/2):ExpandToGrid(4)
    local min = region.CFrame.p - region.Size/2
    local regions = {}
    local start = getAlignedPosition(min)
    local xOffset, yOffset, zOffset = 0,0,0
    local done = false
    repeat
        local pos = start + Vector3.new(xOffset, yOffset, zOffset)
        local sizeVec = region.CFrame.p + region.Size/2 - pos
        local x = math.min(sizeVec.X, processBlockSize)
        local y = math.min(sizeVec.Y, processBlockSize)
        local z = math.min(sizeVec.Z, processBlockSize)
        local chunk = Region3.new(pos, pos + Vector3.new(x, y, z)):ExpandToGrid(4)
        table.insert(regions, chunk)
        zOffset = zOffset + processBlockSize
        if zOffset >= region.Size.Z then
            zOffset = 0
            yOffset = yOffset + processBlockSize
        end
        if yOffset >= region.Size.Y then
            yOffset = 0
            xOffset = xOffset + processBlockSize
        end
        if xOffset >= region.Size.X then
            done = true
        end
    until done
    for i, chunk in ipairs(regions) do
        local changed = false
        local materials, occupancy = Terrain:ReadVoxels(chunk, 4)
        for x = 1, materials.Size.X do
            for y = 1, materials.Size.Y do
                for z = 1, materials.Size.Z do
                    if materials[x][y][z] == WATER then
                        materials[x][y][z] = AIR
                        occupancy[x][y][z] = 0
                        changed = true
                    end
                end
            end
        end
        if changed then
            Terrain:WriteVoxels(chunk, 4, materials, occupancy)
        end
        if i % 4 == 0 then wait() end
    end
end

                                end

                    end
                    for i = 1, 100 do
                        if not mailboxAutofarmEnabled then return end
                        task.wait(0.1)
                    end
                end

        end

FarmSection:addButton("Money Counter Dupe", function()
    local player = game.Players.LocalPlayer
    local machines = workspace:WaitForChild("MoneyCounterMachines"):GetChildren()
    local specialMachine = machines[4]
    local specialPickup = specialMachine and specialMachine:FindFirstChild("Base") and specialMachine.Base:FindFirstChild("Pickup")
    local specialPickupInteraction = specialPickup and specialPickup:FindFirstChild("Interaction") or nil

    local usablePrompts = {}

    for _, machine in ipairs(machines) do
        if machine:FindFirstChild("Base") then
            local pickupPart = machine.Base:FindFirstChild("Pickup")
            if pickupPart and pickupPart:FindFirstChild("Interaction") then
                local pickupPrompt = pickupPart.Interaction
                if pickupPrompt:IsA("ProximityPrompt") and pickupPrompt ~= specialPickupInteraction then
                    pickupPrompt.Enabled = false
                    pickupPrompt.HoldDuration = 0
                    pickupPrompt.Exclusivity = Enum.ProximityPromptExclusivity.AlwaysShow
                end
            end
            local interactPart = machine.Base:FindFirstChild("Interact")
            if interactPart and interactPart:FindFirstChild("Interaction") then
                local interactionPrompt = interactPart.Interaction
                if interactionPrompt:IsA("ProximityPrompt") then
                    interactionPrompt.HoldDuration = 0
                    interactionPrompt.Exclusivity = Enum.ProximityPromptExclusivity.AlwaysShow
                end
                table.insert(usablePrompts, interactPart.Interaction)
            end
        end
    end

    if specialPickupInteraction and specialPickupInteraction:IsA("ProximityPrompt") then
        specialPickupInteraction.HoldDuration = 0
        specialPickupInteraction.Exclusivity = Enum.ProximityPromptExclusivity.AlwaysShow
        table.insert(usablePrompts, specialPickupInteraction)
    end

    local function getNearestPrompt()
        local nearestPrompt = nil
        local shortestDist = math.huge
        local plrPos = player.Character and player.Character:FindFirstChild("HumanoidRootPart") and player.Character.HumanoidRootPart.Position
        if plrPos then
            for _, prompt in ipairs(usablePrompts) do
                if prompt:IsA("ProximityPrompt") and prompt.Parent and prompt.Parent:IsA("BasePart") then
                    local promptPos = prompt.Parent.Position
                    local dist = (promptPos - plrPos).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        nearestPrompt = prompt
                    end
                end
            end
        end
        return nearestPrompt
    end

    task.spawn(function()
        while task.wait(0.15) do
            local nearestPrompt = getNearestPrompt()
            if nearestPrompt and nearestPrompt:IsA("ProximityPrompt") and nearestPrompt.Enabled then
                nearestPrompt.HoldDuration = 0
                nearestPrompt.Exclusivity = Enum.ProximityPromptExclusivity.AlwaysShow
                if fireproximityprompt then
                    pcall(function()
                        fireproximityprompt(nearestPrompt)
                    end)
                elseif game:GetService("VirtualInputManager") then
                    local vim = game:GetService("VirtualInputManager")
                    vim:SendKeyEvent(true, Enum.KeyCode.E, false, game)
                    task.wait(0.05)
                    vim:SendKeyEvent(false, Enum.KeyCode.E, false, game)
                end
            end
        end
    end)
    function pressE()
        local VirtualInputManager = game:GetService("VirtualInputManager")
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
        task.wait(0.1)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
    end
    pressE()
end)

local function teleportAndFireWashingMachinePrompt()
    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local children = workspace:GetChildren()
    local targetObj = children[1819]
    local successfulTeleport = false

    if root and targetObj then
        local washingMachinePrompt = nil
        local subChildren = targetObj:GetChildren()
        local entry = subChildren and subChildren[2]
        if targetObj:IsA("BasePart") then
            root.CFrame = targetObj.CFrame + Vector3.new(0, 2, 0)
            successfulTeleport = true
        elseif targetObj:IsA("Model") then
            local primaryPart = targetObj.PrimaryPart or targetObj:FindFirstChildWhichIsA("BasePart")
            if primaryPart then
                root.CFrame = primaryPart.CFrame + Vector3.new(0, 2, 0)
                successfulTeleport = true
            end
        end

         if entry and entry:FindFirstChild("WashingMachine") then
            local washingMachine = entry.WashingMachine
            if washingMachine:FindFirstChild("Interaction") and washingMachine.Interaction:FindFirstChild("Prompt") then
                washingMachinePrompt = washingMachine.Interaction.Prompt
            end
        end

         if washingMachinePrompt then
            task.spawn(function()
                pcall(function()
                    if washingMachinePrompt:IsA("ProximityPrompt") then
                        washingMachinePrompt.HoldDuration = 0
                        washingMachinePrompt:InputHoldBegin()
                         washingMachinePrompt:InputHoldEnd()
                    elseif washingMachinePrompt.FireServer then
                        washingMachinePrompt:FireServer()
                    else
                         pressE()
                    end
                end)
            end)
        else
             pressE()
        end

    end

     if root and not successfulTeleport then
        root.CFrame = CFrame.new(
            -373.593689, 335.593597, -822.125061,
            -0.325101256, 0, 0.945679247,
            0, 1, 0,
            -0.945679247, 0, -0.325101256
        )
    end
end

local function TELEPORT_LOOP()
